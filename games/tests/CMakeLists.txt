cmake_minimum_required(VERSION 3.21)
project(portal_app_tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS QuickTest WebEngineQuick)
add_executable(app_tests portal-fixture.cpp ../../rust/conquerd-client/src/ui/scheme.cpp)
target_link_libraries(app_tests PRIVATE Qt6::QuickTest Qt6::WebEngineQuick)
get_filename_component(repo_root "../.." ABSOLUTE)
target_compile_definitions(app_tests PRIVATE DEMO_ROOT="${repo_root}")
enable_testing()
add_test(NAME portal_apps COMMAND app_tests -platform offscreen -input "${CMAKE_CURRENT_SOURCE_DIR}" -o -,txt)
set_tests_properties(portal_apps PROPERTIES TIMEOUT 120)
